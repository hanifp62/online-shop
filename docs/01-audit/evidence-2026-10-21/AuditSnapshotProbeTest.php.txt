/**
 * Audit-only characterization: passing assertions CONFIRM the current defects.
 * These are not acceptance tests. Only run through the isolated Amadis launcher.
 * All identities, credentials, provider responses, and backup files are synthetic.
 */
class AuditSnapshotProbeTest extends TestCase
{
    use RefreshDatabase;

    private string $providerKey;

    protected function setUp(): void
    {
        parent::setUp();
        $this->providerKey = bin2hex(random_bytes(32));
        config(['services.midtrans.server_key' => $this->providerKey, 'services.midtrans.is_production' => false]);
        Http::preventStrayRequests();
        Notification::fake();
    }

    private function order(?string $voucher = null): Order
    {
        $category = Category::create(['cat_title' => 'Audit synthetic']);
        $product = Product::create([
            'product_cat' => $category->cat_id, 'product_brand' => 'ZAFIRA',
            'product_title' => 'Audit synthetic', 'product_slug' => 'audit-'.bin2hex(random_bytes(4)),
            'product_price' => '250000.00',
        ]);
        DB::table('products')->where('product_id', $product->product_id)->update(['stock' => 5, 'is_active' => true]);
        $user = User::factory()->create();
        $address = Address::create([
            'user_id' => $user->id, 'label' => 'Synthetic', 'recipient_name' => 'Audit Synthetic',
            'phone' => '081200000000', 'line1' => 'Jalan Uji', 'city' => 'Jakarta',
            'province' => 'DKI', 'postal_code' => '10110',
        ]);
        $this->actingAs($user)->postJson('/cart/items', ['product_id' => $product->product_id, 'quantity' => 1])->assertOk();
        $id = $this->postJson('/checkout', [
            'address_id' => $address->id, 'idempotency_key' => 'audit-'.bin2hex(random_bytes(8)),
            'voucher_code' => $voucher,
        ])->assertCreated()->json('order.id');

        return Order::with('items')->findOrFail($id);
    }

    private function event(Order $order, string $status = 'settlement', string $fraud = 'accept'): array
    {
        $body = [
            'order_id' => (string) $order->id, 'status_code' => '200',
            'gross_amount' => $order->total.'.00', 'transaction_id' => 'audit-tx-'.$order->id,
            'transaction_status' => $status, 'fraud_status' => $fraud, 'currency' => 'IDR',
        ];
        $body['signature_key'] = hash('sha512', $body['order_id'].$body['status_code'].$body['gross_amount'].$this->providerKey);

        return $body;
    }

    public function test_probe_capture_challenge_then_accept_remains_pending(): void
    {
        $order = $this->order();
        $challenge = $this->event($order, 'capture', 'challenge');
        $accept = $this->event($order, 'capture', 'accept');
        Http::fake(['api.sandbox.midtrans.com/v2/*' => Http::sequence()->push($challenge)->push($accept)]);
        $this->postJson('/webhooks/midtrans', $challenge)->assertOk();
        $this->postJson('/webhooks/midtrans', $accept)->assertOk();
        Http::assertSentCount(2);
        $this->assertSame('pending', $order->fresh()->status);
        $this->assertSame('recorded', Payment::sole()->outcome);
    }

    public function test_probe_capture_then_settlement_is_classified_as_nonpayable(): void
    {
        $order = $this->order();
        $capture = $this->event($order, 'capture');
        $settlement = $this->event($order);
        Http::fake(['api.sandbox.midtrans.com/v2/*' => Http::sequence()->push($capture)->push($settlement)]);
        $this->postJson('/webhooks/midtrans', $capture)->assertOk();
        $this->postJson('/webhooks/midtrans', $settlement)->assertOk();
        $this->assertSame('paid', $order->fresh()->status);
        $this->assertSame(['paid', 'order_not_payable'], Payment::orderBy('id')->pluck('outcome')->all());
    }

    public function test_probe_paid_cancellation_has_no_refund_obligation(): void
    {
        $order = $this->order();
        $event = $this->event($order);
        Http::fake(['api.sandbox.midtrans.com/v2/*' => Http::response($event)]);
        $this->postJson('/webhooks/midtrans', $event)->assertOk();
        $this->assertSame('paid', $order->fresh()->status);
        $this->postJson('/orders/'.$order->id.'/cancel')->assertOk();
        $this->assertSame('cancelled', $order->fresh()->status);
        $this->assertSame(5, Product::findOrFail($order->items->first()->product_id)->stock);
        $this->assertDatabaseCount('refunds', 0);
        $this->getJson('/orders/'.$order->id.'/return-eligibility')->assertOk()->assertJsonPath('eligible', false);
    }

    public function test_probe_discounted_full_return_cannot_request_refund(): void
    {
        Voucher::create(['code' => 'AUDIT', 'discount_amount' => 50000, 'quota' => 10, 'used' => 0, 'per_user_limit' => 1, 'is_active' => true]);
        $order = $this->order('AUDIT');
        $this->assertSame(200000, $order->total);
        $event = $this->event($order);
        Http::fake(['api.sandbox.midtrans.com/v2/*' => Http::response($event)]);
        $this->postJson('/webhooks/midtrans', $event)->assertOk();
        foreach (['processing', 'shipped', 'completed'] as $status) {
            $order = app(CheckoutService::class)->transition($order->fresh(), $status);
        }
        $return = $this->postJson('/orders/'.$order->id.'/returns', [
            'reason' => 'damaged',
            'items' => [['order_item_id' => $order->items->first()->id, 'quantity' => 1]],
        ])->assertCreated()->json('return');
        $this->assertSame(250000, $return['requested_refund_amount']);
        $this->postJson('/returns/'.$return['id'].'/refunds', ['reason' => 'Synthetic audit return'])
            ->assertUnprocessable()->assertJsonValidationErrors('amount');
    }

    public function test_probe_reset_url_uses_untrusted_request_host(): void
    {
        $user = User::factory()->create();
        $this->post('https://audit-untrusted.example.test/lupa-kata-sandi', ['email' => $user->email])->assertRedirect();
        Notification::assertSentTo($user, ResetPassword::class, function ($notification) use ($user) {
            $host = parse_url($notification->toMail($user)->actionUrl, PHP_URL_HOST);
            $this->assertSame('audit-untrusted.example.test', $host);

            return true;
        });
    }

    public function test_probe_password_reset_array_email_returns_server_error(): void
    {
        $this->postJson('/lupa-kata-sandi', ['email' => ['synthetic@example.test']])->assertStatus(500);
    }

    public function test_probe_recovery_codes_survive_admin_identity_change_with_web_guard(): void
    {
        $user = User::factory()->create();
        $password = 'Aa1!'.bin2hex(random_bytes(16));
        $adminA = Admin::create(['name' => 'Audit A', 'email' => 'audit-a@example.test', 'password' => $password]);
        $adminB = Admin::create(['name' => 'Audit B', 'email' => 'audit-b@example.test', 'password' => $password]);
        $this->actingAs($user, 'web')->actingAs($adminA, 'admin');
        $this->get('/admin/totp/enrol')->assertOk();
        $secret = Crypt::decryptString(session(AdminTotpEnrolController::PENDING_SECRET));
        $otp = app(Google2FA::class)->getCurrentOtp($secret);
        $this->post('/admin/totp/enrol', ['code' => $otp])->assertRedirect(route('admin.totp.enrol.recovery'));
        $codes = json_decode(Crypt::decryptString(session(AdminTotpEnrolController::FRESH_CODES)), true);
        $this->post('/admin/logout')->assertRedirect();
        $this->assertTrue(Auth::guard('web')->check());
        $this->post('/admin/login', ['email' => $adminB->email, 'password' => $password])->assertRedirect();
        $this->assertSame($adminB->id, Auth::guard('admin')->id());
        $response = $this->get('/admin/totp/enrol/recovery')->assertOk();
        $this->assertTrue(str_contains($response->getContent(), $codes[0]), 'A recovery code from the previous principal was rendered.');
    }

    public function test_probe_zero_backup_retention_deletes_all_synthetic_artifacts(): void
    {
        $dir = storage_path('app/audit-retention-'.bin2hex(random_bytes(6)));
        File::ensureDirectoryExists($dir);
        config(['backup.path' => $dir]);
        try {
            foreach (['20260101', '20260102'] as $date) {
                file_put_contents($dir.'/sqlite-'.$date.'.sqlite.gz', gzencode('synthetic'));
            }
            $service = new DatabaseBackupService('sqlite');
            $this->assertCount(2, $service->backups());
            $this->assertSame(2, $service->prune(0));
            $this->assertSame([], $service->backups());
        } finally {
            File::deleteDirectory($dir);
        }
    }

    public function test_probe_backup_ignores_database_url_target(): void
    {
        $dir = storage_path('app/audit-dump-'.bin2hex(random_bytes(6)));
        config(['backup.path' => $dir, 'database.connections.audit_dump' => [
            'driver' => 'mysql', 'url' => 'mysql://url-host.invalid/expected_database',
            'host' => 'fallback.invalid', 'port' => 3306, 'username' => '', 'password' => '', 'database' => 'fallback_database',
        ]]);
        $seen = [];
        Process::fake(function ($process) use (&$seen) {
            $seen = $process->command;
            foreach ($seen as $arg) {
                if (str_starts_with($arg, '--result-file=')) {
                    file_put_contents(substr($arg, 14), '-- synthetic dump; no database contacted');
                }
            }

            return Process::result();
        });
        try {
            (new DatabaseBackupService('audit_dump'))->backup();
            $this->assertContains('--host=fallback.invalid', $seen);
            $this->assertContains('fallback_database', $seen);
            $this->assertNotContains('expected_database', $seen);
        } finally {
            File::deleteDirectory($dir);
        }
    }
}
